import re
import difflib


def normalize_ada_file(file_path):
    code_lines = []

    with open(file_path, "r", encoding="utf-8") as f:
        for line in f:
            # Remove comments
            line = re.sub(r'--.*$', '', line)

            # Remove leading/trailing spaces
            line = line.strip()

            # Skip empty lines
            if not line:
                continue

            # Skip pragma statements
            if re.match(r'^\s*pragma\b', line, re.IGNORECASE):
                continue

            code_lines.append(line)

    return code_lines


def compare_ada_files(file1, file2):
    code1 = normalize_ada_file(file1)
    code2 = normalize_ada_file(file2)

    if code1 == code2:
        print("Files are identical (ignoring comments and pragmas)")
        return

    print("Differences found:\n")

    diff = difflib.unified_diff(
        code1,
        code2,
        fromfile=file1,
        tofile=file2,
        lineterm=""
    )

    for line in diff:
        print(line)


# Example
compare_ada_files("old.adb", "new.adb")
