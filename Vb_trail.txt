    "/Build_Files"

    MsgBox "Processing Complete"

Function IsExcluded(ByVal pathValue As String) As Boolean

    If InStr(1, pathValue, EXCLUDE_PATH, vbTextCompare) > 0 Then
        IsExcluded = True
    Else
        IsExcluded = False
    End If

End Function


Sub RemoveExcludedRows(ByVal wsSource As Worksheet, _
                       ByVal wsExcluded As Worksheet)

    Dim lastRow As Long
    Dim targetRow As Long
    Dim r As Long
    Dim pathValue As String

    'Find last used row in source sheet
    lastRow = wsSource.Cells(wsSource.Rows.Count, 1).End(xlUp).Row

    'Find next available row in excluded sheet
    targetRow = wsExcluded.Cells(wsExcluded.Rows.Count, 1).End(xlUp).Row + 1

    'Process from bottom to top
    For r = lastRow To 2 Step -1

        pathValue = CStr(wsSource.Cells(r, 1).Value)

        If IsExcluded(pathValue) Then

            'Copy row
            wsSource.Rows(r).Copy _
                Destination:=wsExcluded.Rows(targetRow)

            targetRow = targetRow + 1

            'Delete row from source
            wsSource.Rows(r).Delete

        End If

    Next r

End Sub

Sub CreateFileNameColumn()

    Dim ws As Worksheet
    Dim lastRow As Long
    Dim i As Long
    Dim pathText As String
    Dim fileName As String
    Dim prefix As String
    Dim parts() As String

    Set ws = ActiveSheet

    ' Insert new column C
    ws.Columns("C").Insert Shift:=xlToRight

    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row

    For i = 6 To lastRow

        pathText = Trim(ws.Cells(i, "A").Value)

        If pathText <> "" Then

            parts = Split(pathText, "/")
            fileName = parts(UBound(parts))

            prefix = ""

            If InStr(1, LCase(pathText), "control") > 0 Then
                prefix = "Control#"
            ElseIf InStr(1, LCase(pathText), "common") > 0 Then
                prefix = "Common#"
            ElseIf InStr(1, LCase(pathText), "safety") > 0 Then
                prefix = "Safety#"
            End If

            ws.Cells(i, "C").Value = prefix & fileName

        End If

    Next i

End Sub

Sub ApplyVLookup()

    Dim ws As Worksheet
    Dim lookupWs As Worksheet
    Dim lastRow As Long
    Dim i As Long

    Set ws = Sheets("Sheet1")          ' destination sheet
    Set lookupWs = Sheets("Lookup")    ' lookup sheet

    lastRow = ws.Cells(ws.Rows.Count, "B").End(xlUp).Row

    For i = 6 To lastRow

        ws.Cells(i, "H").Formula = _
            "=IFERROR(VLOOKUP(B" & i & ",'" & lookupWs.Name & _
            "'!B:C,2,FALSE),"""")"

    Next i

    ' Convert formulas to values (optional)
    ws.Range("H6:H" & lastRow).Value = ws.Range("H6:H" & lastRow).Value

End Sub

Sub CreateUniqueFileList()

    Dim wsGR5 As Worksheet
    Dim wsGR6 As Worksheet
    Dim wsResult As Worksheet
    Dim dict As Object
    Dim lastRow As Long
    Dim i As Long
    Dim key As String
    Dim outRow As Long

    Set wsGR5 = Sheets("GR5")
    Set wsGR6 = Sheets("GR6")
    Set wsResult = Sheets("Result")

    Set dict = CreateObject("Scripting.Dictionary")

    'Clear result sheet
    wsResult.Cells.Clear

    'Heading
    wsResult.Range("A1").Value = "FileName"

    'Read GR5
    lastRow = wsGR5.Cells(wsGR5.Rows.Count, "B").End(xlUp).Row

    For i = 6 To lastRow
        key = Trim(wsGR5.Cells(i, "B").Value)
        If key <> "" Then
            If Not dict.Exists(key) Then
                dict.Add key, ""
            End If
        End If
    Next i

    'Read GR6
    lastRow = wsGR6.Cells(wsGR6.Rows.Count, "B").End(xlUp).Row

    For i = 6 To lastRow
        key = Trim(wsGR6.Cells(i, "B").Value)
        If key <> "" Then
            If Not dict.Exists(key) Then
                dict.Add key, ""
            End If
        End If
    Next i

    'Write unique values to Result sheet
    outRow = 2

    For Each key In dict.Keys
        wsResult.Cells(outRow, "A").Value = key
        outRow = outRow + 1
    Next key

    MsgBox dict.Count & " unique filenames copied."

End Sub

Sub CompareColumns()

    Dim ws As Worksheet
    Dim lastRow As Long
    Dim i As Long

    Set ws = Sheets("Result")

    lastRow = ws.Cells(ws.Rows.Count, "B").End(xlUp).Row

    ws.Range("D1").Value = "Status"

    For i = 2 To lastRow

        If Trim(ws.Cells(i, "B").Value) = Trim(ws.Cells(i, "C").Value) Then

            ws.Cells(i, "D").Value = "Same"
            ws.Cells(i, "D").Interior.Color = RGB(0, 255, 0)   'Green

        Else

            ws.Cells(i, "D").Value = "Mismatch"
            ws.Cells(i, "D").Interior.Color = RGB(255, 0, 0)   'Red
            ws.Cells(i, "D").Font.Color = RGB(255, 255, 255)   'White text

        End If

    Next i

End Sub

Public Function ImportTextFile(ws As Worksheet, _
                               FilePath As String, _
                               DestCell As String) As Boolean
    On Error GoTo ErrHandler

    With ws

        'Format Column B as Text
        .Columns("B:B").NumberFormat = "@"

        With .QueryTables.Add( _
            Connection:="TEXT;" & FilePath, _
            Destination:=.Range(DestCell))

            .TextFileParseType = xlDelimited
            .TextFileTabDelimiter = True

            'Column 1 = General
            'Column 2 = Text
            .TextFileColumnDataTypes = Array(1, 2)

            .Refresh BackgroundQuery:=False
            .Delete

        End With

    End With

    ImportTextFile = True
    Exit Function

ErrHandler:
    ImportTextFile = False
End Function

Sub RemoveMatchingRowsFast()

    Dim ws As Worksheet
    Dim lookupWs As Worksheet
    Dim dict As Object
    Dim lastRow As Long
    Dim lookupLastRow As Long
    Dim i As Long
    Dim key As Variant
    Dim txt As String

    Set ws = Worksheets("Data")
    Set lookupWs = Worksheets("Lookup")

    Set dict = CreateObject("Scripting.Dictionary")

    lookupLastRow = lookupWs.Cells(lookupWs.Rows.Count, "A").End(xlUp).Row

    For i = 1 To lookupLastRow
        dict(UCase(Trim(lookupWs.Cells(i, "A").Value))) = True
    Next i

    lastRow = ws.Cells(ws.Rows.Count, "B").End(xlUp).Row

    For i = 2 To lastRow

        txt = UCase(ws.Cells(i, "B").Value)

        For Each key In dict.Keys
            If InStr(txt, key) > 0 Then
                ws.Cells(i, "Z").Value = "DELETE"
                Exit For
            End If
        Next key

    Next i

    ws.Range("A1:Z" & lastRow).AutoFilter Field:=26, Criteria1:="DELETE"

    ws.Range("A2:Z" & lastRow).SpecialCells(xlCellTypeVisible).EntireRow.Delete

    ws.AutoFilterMode = False

End Sub

Sub GetTxtFilesRecursive()

    Dim fd As FileDialog
    Dim FolderPath As String

    Set fd = Application.FileDialog(msoFileDialogFolderPicker)

    If fd.Show <> -1 Then Exit Sub

    FolderPath = fd.SelectedItems(1)
    
    GetFiles FolderPath

End Sub

Sub GetFiles(ByVal FolderPath As String)

    Dim fso As Object, fld As Object, sf As Object, fil As Object
    Static R As Long

    If R = 0 Then R = 1

    Set fso = CreateObject("Scripting.FileSystemObject")
    Set fld = fso.GetFolder(FolderPath)

    For Each fil In fld.Files
        If LCase(fso.GetExtensionName(fil.Name)) = "txt" Then
            Cells(R, 1).Value = fil.Path
            R = R + 1
        End If
    Next fil

    For Each sf In fld.SubFolders
        GetFiles sf.Path
    Next sf

End Sub

import re
import csv

input_file = "spark_report.txt"
output_file = "spark_report.csv"

current_file = ""
current_category = ""

with open(output_file, "w", newline="", encoding="utf-8") as csvfile:

    writer = csv.writer(csvfile)

    writer.writerow([
        "Category",
        "File",
        "Occurrence",
        "Type",
        "Code",
        "Message"
    ])

    with open(input_file, "r", encoding="utf-8", errors="ignore") as infile:

        for line in infile:

            line = line.rstrip()

            # Capture section headers
            if "Warnings local to files being SPARKed" in line:
                current_category = "Warnings"
                continue

            elif "Flow warnings local to files being SPARKed" in line:
                current_category = "Flow Warnings"
                continue

            elif "Errors local to files being SPARKed" in line:
                current_category = "Errors"
                continue

            elif "Flow errors local to files being SPARKed" in line:
                current_category = "Flow Errors"
                continue

            # Capture file path
            file_match = re.search(
                r'([A-Z]:\\.*\.(?:LSB|LSS))',
                line,
                re.I
            )

            if file_match:
                current_file = file_match.group(1)
                continue

            # Capture issue lines
            issue_match = re.search(
                r'[\*\?\-!]+\s*\(\s*(\d+)\s*\)\s*'
                r'(Warning|Flow Error|Semantic Error)'
                r'\s*:\s*(\d+)'
                r'\s*:\s*(.*)',
                line
            )

            if issue_match:

                writer.writerow([
                    current_category,
                    current_file,
                    issue_match.group(1),
                    issue_match.group(2),
                    issue_match.group(3),
                    issue_match.group(4).strip()
                ])

print("Completed")
