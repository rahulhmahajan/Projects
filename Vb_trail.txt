Sub RemoveExcludedRows(wsSource As Worksheet, _
                       wsExcluded As Worksheet, _
                       excludeList As Variant)

    Dim lastRow As Long
    Dim targetRow As Long
    Dim r As Long
    Dim item As Variant
    Dim pathValue As String
    Dim isExcluded As Boolean

    lastRow = wsSource.Cells(wsSource.Rows.Count, 1).End(xlUp).Row

    targetRow = wsExcluded.Cells(wsExcluded.Rows.Count, 1).End(xlUp).Row + 1

    'Loop from bottom to top
    For r = lastRow To 2 Step -1

        pathValue = Trim(wsSource.Cells(r, 1).Value)

        isExcluded = False

        For Each item In excludeList

            If StrComp(pathValue, Trim(item), vbTextCompare) = 0 Then

                isExcluded = True
                Exit For

            End If

        Next item

        If isExcluded Then

            wsSource.Rows(r).Copy wsExcluded.Rows(targetRow)

            targetRow = targetRow + 1

            wsSource.Rows(r).Delete

        End If

    Next r

End Sub
