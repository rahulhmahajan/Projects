html = """
<html>
<head>
<style>
body {font-family: Arial;}
h2 {background:#ddd;padding:8px;}
</style>
</head>
<body>
<h1>Ada Comparison Report</h1>
"""

for file_name in common_files:

    code1 = normalize_ada_file(gr5_path)
    code2 = normalize_ada_file(gr6_path)

    diff_html = difflib.HtmlDiff().make_table(
        code1,
        code2,
        fromdesc="GR5",
        todesc="GR6",
        context=True
    )

    html += f"<h2>{file_name}</h2>"
    html += diff_html

html += "</body></html>"

with open("Ada_Report.html", "w", encoding="utf-8") as f:
    f.write(html)
